## WorkAgentHarness 仿写顺序
1. 写共享核心 `RunUseCase`：验证后输入→创建 run→调用 Agent→写唯一终态。HTTP/CLI 仅做输入输出适配。
2. 定义 run 状态机与 cancellation token，从入口透传到 Agent/工具；明确每个耗时边界怎样响应取消。
3. 先实现 CLI 调同一个核心用例，再实现 HTTP adapter；不在各入口重复业务逻辑。
4. 为 App Server 写协议 schema、request id、错误、超时和退出清理，同样调用核心 use case。
5. 若任务需脱离请求协程运行，定义 owner、并发上限、进程退出等待/终止和孤儿恢复策略。
6. 所有成功/异常/取消/关闭路径都执行幂等资源清理，只写一次终态。

## 分层练习与出口
L1 启动/完成/取消一个 fake run；L2 CLI 与 App Server 各跑一次；L3 完整 run 状态、stdio RPC、桥接与平台适配；L4 验证断连、重复启动、取消传播、退出清理。注入正常完成、取消、断连、重复请求、SIGTERM、RPC 超时，检查终态、取消到达和资源释放；交付生命周期图与跨入口一致性记录。

## 跟着写代码：所有入口共用一个 RunService

本章先写同步的单进程版本，把 run 生命周期从 CLI 循环中拿出来。以后增加 HTTP 和 stdio RPC 时，入口只负责解析/返回协议；都调用同一个 `RunService`。

### 1. 定义运行状态和 Repository：`persistence/run_repository.py`

先确保第 4 章的 `runs` 表已加入 `SCHEMA_SQL`，再用 repository 约束终态只从 `running` 转出：

```python
import sqlite3


class RunRepository:
    """封装 run 表 SQL；不调用 Agent，也不处理 CLI 输出。"""

    def __init__(self, conn: sqlite3.Connection) -> None:
        self._conn = conn

    def start(self, run_id: str, session_id: str, owner_id: str, now: str) -> None:
        self._conn.execute(
            "INSERT INTO runs(id, session_id, owner_id, status, created_at, updated_at) "
            "VALUES (?, ?, ?, 'running', ?, ?)",
            (run_id, session_id, owner_id, now, now),
        )
        self._conn.commit()

    def finish(self, run_id: str, status: str, now: str, error_code: str = "") -> None:
        """只允许 running 转到终态；重复完成或迟到更新应显式失败。"""
        if status not in {"completed", "failed", "cancelled"}:
            raise ValueError(f"非法 run 终态: {status}")
        cursor = self._conn.execute(
            "UPDATE runs SET status = ?, error_code = ?, updated_at = ? "
            "WHERE id = ? AND status = 'running'",
            (status, error_code, now, run_id),
        )
        self._conn.commit()
        if cursor.rowcount != 1:
            raise RuntimeError(f"run 不存在或已经结束: {run_id}")

    def complete(self, run_id: str, now: str) -> None:
        self.finish(run_id, "completed", now)

    def cancel(self, run_id: str, now: str) -> None:
        self.finish(run_id, "cancelled", now)

    def fail(self, run_id: str, error_code: str, now: str) -> None:
        self.finish(run_id, "failed", now, error_code)
```

数据库状态条件 `WHERE status = 'running'` 是幂等保护的一部分：两个并发结束回调不能都把同一 run 更新成终态。

### 2. 表示一次运行：`execution/run_service.py`

```python
from dataclasses import dataclass
from datetime import UTC, datetime
from uuid import uuid4


@dataclass(frozen=True)
class RunResult:
    run_id: str                         # 调用者可查询/关联的运行标识
    status: str                         # completed / failed / cancelled
    answer: str = ""
    error_code: str = ""                # 不把异常或密钥直接暴露给 UI


def _now() -> str:
    return datetime.now(UTC).isoformat()  # 所有运行时间采用 UTC


class RunService:
    def __init__(self, agent, sessions, runs) -> None:
        self._agent = agent              # 已构造的 Agent，依赖由入口装配
        self._sessions = sessions        # 业务消息 repository
        self._runs = runs                # run 状态 repository

    def execute(self, owner_id: str, session_id: str, text: str) -> RunResult:
        run_id = uuid4().hex             # 每次尝试单独编号；重试不复用 id
        self._runs.start(run_id, session_id, owner_id, _now())
        self._sessions.add_message(session_id, "user", text)
        try:
            result = self._agent.invoke(
                {"messages": [("user", text)]},
                {"configurable": {"thread_id": session_id}},
            )
            answer = str(result["messages"][-1].content)
            self._sessions.add_message(session_id, "assistant", answer)
            self._runs.complete(run_id, _now())
            return RunResult(run_id=run_id, status="completed", answer=answer)
        except KeyboardInterrupt:
            self._runs.cancel(run_id, _now())  # CLI 中断先形成明确终态
            raise                                # 同时把中断交还给入口处理
        except Exception as exc:
            # 记录稳定错误类别；不要把可能含敏感请求内容的 str(exc) 原样存库。
            code = type(exc).__name__
            self._runs.fail(run_id, code, _now())
            return RunResult(run_id=run_id, status="failed", error_code=code)
```

这里展示职责，不是可直接运行的完整类：`runs.start/complete/cancel/fail` 对应下一步要实现的 repository 方法；消息和 run 的原子事务也要在本章 L2 扩展。

### 3. CLI 只做装配和交互：`cli/main.py`

```python
import argparse
import uuid


def main() -> None:
    parser = argparse.ArgumentParser(description="WorkAgentHarness 对话")
    parser.add_argument("--thread", help="继续已有 session/thread")
    parser.add_argument("--once", metavar="TEXT", help="只发送一条消息并退出")
    args = parser.parse_args()

    # 所有依赖在入口装配；RunService 不自己读配置或开数据库。
    chat_config = load_chat_model_config("config.yaml")
    db_path = "data/workagentharness.db"  # 后续章节改由应用配置提供
    db = open_database(db_path)
    checkpoint_conn, saver = open_checkpointer(db_path)
    model = create_chat_model(chat_config)
    agent = make_lead_agent(model, checkpointer=saver)
    sessions = SessionRepository(db)
    runs = RunRepository(db)
    service = RunService(agent, sessions, runs)

    session_id = args.thread or uuid.uuid4().hex  # 新会话使用随机 id
    sessions.create(session_id, owner_id="local-user")
    try:
        text = args.once if args.once is not None else input("你> ")
        result = service.execute("local-user", session_id, text)
        if result.status == "completed":
            print(f"助手> {result.answer}")
        else:
            print(f"运行失败: {result.error_code} (run={result.run_id})")
    finally:
        checkpoint_conn.close()  # 持久连接由装配层负责释放
        db.close()


if __name__ == "__main__":
    main()  # `python -m workagentharness.cli.main` 从这里启动
```

### 4. 为什么不要把 `KeyboardInterrupt` 当完整取消

上面的代码只适合单进程 CLI 学习：主线程收到 Ctrl+C 后，run 标成 `cancelled`，但真实 HTTP 请求、子代理、工具和后台进程未必都停止。L3/L4 必须继续把取消 token 传过 Agent 与每个耗时边界，并在 `finally` 清理资源；将来 HTTP disconnect 也要触发同一取消机制。

### 5. 本章按顺序扩展

1. L1：内存 RunRepository + fake model，跑通 completed/failed/cancelled 三种终态。
2. L2：SQLite RunRepository、CLI `--thread` 和 `--once`，复用第 4 章 checkpoint。
3. L3：stdio RPC/App Server adapter；协议解析完成后调用同一个 `RunService.execute`。
4. L4：加入可传播 cancellation token、幂等清理、断连检测、并发限制和进程退出等待。

完成后，CLI 和 App Server 对同一输入应产生同一 run 状态与会话记录；入口差异只存在于协议适配层。

### 6. 增加本地 stdio App Server：`app_server/stdio_rpc.py`

本篇先实现一行一个 JSON-RPC 请求/响应的教学子集。完整协议还要覆盖流式通知、取消、并发 run、宿主桥接和关闭语义；本节不把子集说成 EvoFlow 的完整复刻。

```python
def serve_stdio(service, sessions, owner_id="local-user", stdin=None, stdout=None):
    """协议层读写 JSON；业务执行仍统一委托给 RunService。"""
    source = stdin or sys.stdin
    sink = stdout or sys.stdout
    for line in source:
        request = json.loads(line)  # 一行一个 JSON-RPC 请求
        request_id = request.get("id")
        method = request.get("method")
        params = request.get("params", {})

        if method == "initialize":
            result = {"serverInfo": {"name": "WorkAgentHarness", "version": "0.1.0"}}
        elif method == "server/ping":
            result = {"ok": True}
        elif method == "turn/start":
            session_id = params.get("sessionId") or uuid4().hex
            sessions.create(session_id, owner_id)
            run = service.execute(owner_id, session_id, params["text"])
            result = {"sessionId": session_id, "runId": run.run_id,
                      "status": run.status, "answer": run.answer}
        else:
            # JSON-RPC 错误放在顶层 error 字段，不混进业务 result。
            if "id" in request:  # notification 没有响应，包括未知方法
                sink.write(json.dumps({"jsonrpc": "2.0", "id": request_id,
                                       "error": {"code": -32601, "message": "未知方法"}},
                                      ensure_ascii=False) + "\n")
            continue

        if "id" in request:  # notification 不需要响应
            sink.write(json.dumps({"jsonrpc": "2.0", "id": request_id,
                                   "result": result}, ensure_ascii=False) + "\n")
```

上面是协议流程的教学摘录；可运行版本还处理无效 JSON、无效参数、未知方法和内部错误，并在 [stdio_rpc.py](../../../../backend/packages/harness/workagentharness/app_server/stdio_rpc.py) 中维护。它支持 `initialize`、`server/ping`、`turn/start`，尚未支持取消、流事件或完整宿主协议。
